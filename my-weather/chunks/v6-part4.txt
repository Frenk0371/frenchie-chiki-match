function briefCondition(hs){
  if(!hs.length)return'';
  const maxPop=Math.max(...hs.map(h=>Number(h.pop)||0));
  const rainMm=hs.reduce((s,h)=>s+(Number(h.precip)||0),0);
  const storm=hs.filter(h=>Number(h.code)>=95&&Number(h.pop)>=40).length;
  const snow=hs.filter(h=>Number(h.code)>=71&&Number(h.code)<=77).length;
  const fog=hs.filter(h=>Number(h.code)>=45&&Number(h.code)<=48).length;
  const wet=hs.filter(h=>Number(h.code)>=51&&(Number(h.pop)>=35||(Number(h.precip)||0)>=.2)).length;
  const cloud=hs.map(h=>Number(h.cloud)).filter(Number.isFinite);
  const avgCloud=cloud.length?cloud.reduce((a,b)=>a+b,0)/cloud.length:50;
  if(storm)return'con possibili temporali';
  if(snow)return'con possibili nevicate';
  if(wet>=Math.max(1,Math.ceil(hs.length*.25))&&(maxPop>=35||rainMm>=.5))return'con possibili rovesci';
  if(fog>=Math.max(1,Math.ceil(hs.length*.35)))return'con nebbia';
  if(avgCloud>=78)return'nuvoloso';
  if(avgCloud>=38)return'con qualche nube';
  return'sereno';
}
function todayBrief(hours,daily){
  if(!hours?.length)return{title:'Oggi in breve',text:'Dettagli non disponibili.'};
  const today=hours[0].time.slice(0,10);
  const hs=hours.filter(h=>h.time.slice(0,10)===today);
  const groups=[
    ['Mattina',hs.filter(h=>+h.time.slice(11,13)>=6&&+h.time.slice(11,13)<12)],
    ['Pomeriggio',hs.filter(h=>+h.time.slice(11,13)>=12&&+h.time.slice(11,13)<18)],
    ['Sera',hs.filter(h=>+h.time.slice(11,13)>=18&&+h.time.slice(11,13)<=23)]
  ].filter(x=>x[1].length).map(([name,g])=>({name,cond:briefCondition(g)}));
  let text='';
  if(!groups.length){
    text='Giornata quasi conclusa.';
  }else{
    const chunks=[];
    for(let i=0;i<groups.length;){
      let j=i;
      while(j+1<groups.length&&groups[j+1].cond===groups[i].cond)j++;
      const names=groups.slice(i,j+1).map(x=>x.name.toLowerCase());
      let when;
      if(names.length===1)when=(i===0?groups[i].name:names[0]);
      else if(names.length===2)when=(i===0?groups[i].name:names[0])+' e '+names[1];
      else when=(i===0?groups[i].name:names[0])+', '+names.slice(1,-1).join(', ')+' e '+names[names.length-1];
      chunks.push(when+' '+groups[i].cond);
      i=j+1;
    }
    text=chunks.join(', ')+'.';
  }
  if(daily&&Number.isFinite(Number(daily.max)))text+=' Massima '+Math.round(daily.max)+'°.';
  return{title:'Oggi in breve',text};
}

  if(first<0)return{wet:false,heading:'Quando pioverà?',title:'Nessuna pioggia significativa',detail:'Nelle prossime 24 ore'};
  let end=first,lastWet=first,gap=0;
  for(let i=first+1;i<hs.length;i++){
    if(significant(hs[i])){lastWet=i;end=i;gap=0}
    else if(gap<1){gap++;end=i}
    else break;
  }
  end=lastWet;
  const seg=hs.slice(first,end+1),maxPop=Math.max(...seg.map(h=>Number(h.pop)||0));
  const total=seg.reduce((s,h)=>s+(Number(h.precip)||0),0);
  const snowHours=seg.filter(h=>Number(h.code)>=71&&Number(h.code)<=77&&significant(h)).length;
  const rainHours=seg.filter(h=>Number(h.code)>=51&&!(Number(h.code)>=71&&Number(h.code)<=77)&&significant(h)).length;
  const isSnow=snowHours>rainHours&&snowHours>0;
  const mixed=snowHours>0&&rainHours>0&&snowHours===rainHours;
  const refDay=hs[0].time.slice(0,10),startDay=hs[first].time.slice(0,10);
  const diff=Math.round((new Date(startDay+'T12:00')-new Date(refDay+'T12:00'))/86400000);
  const dayWord=diff===0?'oggi':diff===1?'domani':new Intl.DateTimeFormat('it-IT',{weekday:'long'}).format(new Date(startDay+'T12:00'));
  const t0=hs[first].time.slice(11,16),t1=hs[end].time.slice(11,16);
  let when=first===0?'Adesso':(first===end?dayWord+' intorno alle '+t0:dayWord+' dalle '+t0+' alle '+t1);
  let heading,kind;
  if(isSnow){
    heading='Quando nevicherà?';
    kind=maxPop>=70||total>=3?'Neve probabile':'Possibili nevicate';
  }else if(mixed){
    heading='Precipitazioni';
    kind=maxPop>=70||total>=3?'Pioggia e neve probabili':'Possibili pioggia e neve';
  }else{
    heading='Quando pioverà?';
    kind=maxPop>=70||total>=3?'Pioggia probabile':'Possibili rovesci';
  }
  return{wet:true,heading,title:kind+' · '+when,detail:'Picco '+Math.round(maxPop)+'% · '+fmtMm(total)+' nel periodo'};
}
